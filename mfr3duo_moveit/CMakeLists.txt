target_compile_options(mfr3duo_moveit PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(mfr3duo_moveit PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(mfr3duo_moveit rclcpp rclcpp_action tf2_ros geometry_msgs moveit_msgs rcl_interfaces sensor_msgs tf2_msgs controller_manager_msgs moveit_core moveit_ros_planning control_msgs action_msgs std_msgs)

install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_mfr3duo_moveit HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp rclcpp_action tf2_ros geometry_msgs moveit_msgs rcl_interfaces sensor_msgs tf2_msgs controller_manager_msgs moveit_core moveit_ros_planning ompl control_msgs action_msgs std_msgs)
