cmake_minimum_required(VERSION 3.16)
project(mfr3duo_robot VERSION 0.1.0 LANGUAGES CXX)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(mfr3duo_moveit REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(mfr3duo_control REQUIRED)
find_package(mfr3duo_nav REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(yaml-cpp REQUIRED)
add_library(mfr3duo_robot_observer OBJECT src/grasp_observer.cpp src/grasp_recovery.cpp)
target_compile_features(mfr3duo_robot_observer PRIVATE cxx_std_17)
target_include_directories(mfr3duo_robot_observer PUBLIC include)
set_target_properties(mfr3duo_robot_observer PROPERTIES POSITION_INDEPENDENT_CODE ON)
ament_target_dependencies(mfr3duo_robot_observer rclcpp geometry_msgs sensor_msgs ament_index_cpp tf2_ros)
target_link_libraries(mfr3duo_robot_observer mfr3duo_moveit::mfr3duo_moveit mfr3duo_control::mfr3duo_control Eigen3::Eigen yaml-cpp)
add_library(mfr3duo_robot SHARED src/task.cpp src/robot.cpp)
add_library(mfr3duo_robot::mfr3duo_robot ALIAS mfr3duo_robot)
target_compile_features(mfr3duo_robot PUBLIC cxx_std_17)
target_include_directories(mfr3duo_robot PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE src)
ament_target_dependencies(mfr3duo_robot PUBLIC rclcpp geometry_msgs)
target_link_libraries(mfr3duo_robot PRIVATE mfr3duo_robot_observer mfr3duo_control::mfr3duo_control mfr3duo_moveit::mfr3duo_moveit mfr3duo_nav::mfr3duo_nav Eigen3::Eigen yaml-cpp ament_index_cpp::ament_index_cpp tf2_ros::tf2_ros)
target_compile_options(mfr3duo_robot PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS mfr3duo_robot EXPORT export_mfr3duo_robot LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_mfr3duo_robot HAS_LIBRARY_TARGET NAMESPACE mfr3duo_robot::)
ament_export_dependencies(rclcpp geometry_msgs)
add_executable(task_demo tools/task_demo.cpp)
target_link_libraries(task_demo mfr3duo_robot::mfr3duo_robot)
install(TARGETS task_demo DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch maps DESTINATION share/${PROJECT_NAME})
